# MongoDB Diagnostic Commands
# Copy and paste these into your SSH session

# 1. Check detailed MongoDB logs
echo "========== MongoDB Error Logs =========="
sudo tail -n 100 /var/log/mongodb/mongod.log

# 2. Check MongoDB config file
echo ""
echo "========== MongoDB Config =========="
sudo cat /etc/mongod.conf

# 3. Check data directory permissions
echo ""
echo "========== Data Directory Permissions =========="
sudo ls -lah /var/lib/mongodb/

# 4. Check disk space
echo ""
echo "========== Disk Space =========="
df -h

# 5. Check if MongoDB port is in use
echo ""
echo "========== Port 27017 Status =========="
sudo netstat -tuln | grep 27017

# 6. Try to start MongoDB manually to see error
echo ""
echo "========== Manual Start Attempt =========="
sudo -u mongodb mongod --config /etc/mongod.conf

# MongoDB Fix Commands v2 - Based on exit code 14 error
# This usually means config file issue or permission problem

# 1. Stop any existing MongoDB processes
sudo systemctl stop mongod
sudo pkill -9 mongod
sleep 2

# 2. Remove any lock files
sudo rm -f /var/lib/mongodb/mongod.lock
sudo rm -f /tmp/mongodb-27017.sock

# 3. Fix all permissions thoroughly
sudo chown -R mongodb:mongodb /var/lib/mongodb
sudo chown -R mongodb:mongodb /var/log/mongodb
sudo chown mongodb:mongodb /etc/mongod.conf
sudo chmod 755 /var/lib/mongodb
sudo chmod 755 /var/log/mongodb
sudo chmod 644 /etc/mongod.conf

# 4. Backup and recreate config file
sudo cp /etc/mongod.conf /etc/mongod.conf.broken
sudo tee /etc/mongod.conf > /dev/null <<EOF
# mongod.conf

# for documentation of all options, see:
#   http://docs.mongodb.org/manual/reference/configuration-options/

# Where and how to store data.
storage:
  dbPath: /var/lib/mongodb
  journal:
    enabled: true

# where to write logging data.
systemLog:
  destination: file
  logAppend: true
  path: /var/log/mongodb/mongod.log

# network interfaces
net:
  port: 27017
  bindIp: 127.0.0.1

# how the process runs
processManagement:
  timeZoneInfo: /usr/share/zoneinfo

# security
security:
  authorization: enabled
EOF

# 5. Fix config ownership again
sudo chown mongodb:mongodb /etc/mongod.conf
sudo chmod 644 /etc/mongod.conf

# 6. Start MongoDB
sudo systemctl start mongod
sleep 3

# 7. Check status
sudo systemctl status mongod --no-pager
